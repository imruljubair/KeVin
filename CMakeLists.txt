cmake_minimum_required(VERSION 3.20)
project(kavin LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_METAL OFF CACHE BOOL "" FORCE)
set(LLAMA_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/.deps/llama.cpp" CACHE PATH "Pinned llama.cpp checkout")
add_subdirectory(${LLAMA_SOURCE_DIR} llama)
add_executable(kavin-engine engine/main.cpp)
target_include_directories(kavin-engine PRIVATE ${LLAMA_SOURCE_DIR}/vendor ${LLAMA_SOURCE_DIR}/src)
target_link_libraries(kavin-engine PRIVATE llama ggml)
set_target_properties(kavin-engine PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
